cmake_minimum_required(VERSION 3.16)
project(MHACAfterimage LANGUAGES CXX)
add_executable(afterimage_lab lab.cpp)
target_compile_features(afterimage_lab PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(afterimage_lab PRIVATE /W4 /WX)
else()
  target_compile_options(afterimage_lab PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME afterimage_native COMMAND afterimage_lab)
