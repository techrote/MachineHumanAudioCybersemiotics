cmake_minimum_required(VERSION 3.25)
project(MHACOrbitLantern LANGUAGES CXX)
add_executable(orbit_lab lab.cpp)
target_compile_features(orbit_lab PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(orbit_lab PRIVATE /W4 /WX)
else()
  target_compile_options(orbit_lab PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME orbit_native COMMAND orbit_lab)
find_package(Threads REQUIRED)
add_executable(orbit_host_test host_test.cpp)
target_compile_features(orbit_host_test PRIVATE cxx_std_17)
target_link_libraries(orbit_host_test PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(orbit_host_test PRIVATE /W4 /WX)
else()
  target_compile_options(orbit_host_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME orbit_host COMMAND orbit_host_test)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(VST3.cmake)
