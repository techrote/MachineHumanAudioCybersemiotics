option(MHAC_BUILD_VST3 "Build the native plugin (requires locked SDK checkout)" OFF)
if(MHAC_BUILD_VST3)
    set(MHAC_VST3_SDK "" CACHE PATH "Clean recursive VST3 SDK 3.8.0 checkout")
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" rev-parse HEAD OUTPUT_VARIABLE sdk_head OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT sdk_head STREQUAL "9fad9770f2ae8542ab1a548a68c1ad1ac690abe0")
        message(FATAL_ERROR "SDK must be the exact official 3.8.0 build 66 commit")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" submodule status --recursive OUTPUT_VARIABLE submodules RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT submodules OR submodules MATCHES "(^|\n)[+-U]")
        message(FATAL_ERROR "Initialize all recursive submodules to recorded gitlinks")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" status --porcelain --untracked-files=all --ignore-submodules=none OUTPUT_VARIABLE dirty RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR dirty)
        message(FATAL_ERROR "SDK and recursive submodules must be clean")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/base" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "3d2e82f8e6bff59c1d8b7a27491a29c2286b5206")
        message(FATAL_ERROR "Incorrect SDK dependency: base")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/cmake" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "de6e54eeaaab35b7145f5c32c279b5e892146e04")
        message(FATAL_ERROR "Incorrect SDK dependency: cmake")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/doc" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "6d4737c9e70750056e731d88d49aa06eefc8a1a4")
        message(FATAL_ERROR "Incorrect SDK dependency: doc")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/pluginterfaces" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "31d6eeba6daaa3e2a8bfbe3e7a90ca0b7fbfbc1c")
        message(FATAL_ERROR "Incorrect SDK dependency: pluginterfaces")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/public.sdk" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "a3911a4615dabbfdfd9d181ee26b05c70c289a95")
        message(FATAL_ERROR "Incorrect SDK dependency: public.sdk")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/tutorials" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "33b73dfbb87f3fde3bce8c0a10cae934dc66ad34")
        message(FATAL_ERROR "Incorrect SDK dependency: tutorials")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/vstgui4" rev-parse HEAD OUTPUT_VARIABLE pinned OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT pinned STREQUAL "76823bdbe286e4bdb9f79ab8986af5ce7202336c")
        message(FATAL_ERROR "Incorrect SDK dependency: vstgui4")
    endif()
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
    add_subdirectory("${MHAC_VST3_SDK}" "${CMAKE_BINARY_DIR}/sdk")
    # SDK macros refer to SDK_ROOT/public SDK outside the SDK subdirectory.
    set(SDK_ROOT "${MHAC_VST3_SDK}")
    set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
    smtg_setup_platform_toolset()
    smtg_setup_symbol_visibility()

    target_include_directories(MHACRestBus PRIVATE "${SDK_ROOT}")

    if(APPLE)
        smtg_target_set_bundle(MHACRestBus BUNDLE_IDENTIFIER "org.mhac.rest-bus" COMPANY_NAME "MHAC")
    endif()
    smtg_target_add_plugin_resources(MHACRestBus RESOURCES "${SDK_ROOT}/LICENSE.txt")
endif()
