cmake_minimum_required(VERSION 3.16)
project(MHACCounterfactualResonator VERSION 0.1.0 LANGUAGES CXX)
add_executable(counterfactual_lab lab.cpp)
target_compile_features(counterfactual_lab PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(counterfactual_lab PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(counterfactual_lab PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME counterfactual_native COMMAND counterfactual_lab)
add_executable(counterfactual_probe probe.cpp)
target_compile_features(counterfactual_probe PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(counterfactual_probe PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(counterfactual_probe PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
add_test(NAME counterfactual_probe COMMAND counterfactual_probe)
find_package(Threads REQUIRED)
add_executable(counterfactual_host_test host_test.cpp)
target_compile_features(counterfactual_host_test PRIVATE cxx_std_17)
target_link_libraries(counterfactual_host_test PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(counterfactual_host_test PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(counterfactual_host_test PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
add_test(NAME counterfactual_host COMMAND counterfactual_host_test)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(VST3.cmake)

add_executable(counterfactual_presets preset_bank.cpp)
target_compile_features(counterfactual_presets PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(counterfactual_presets PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(counterfactual_presets PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
add_test(NAME counterfactual_presets COMMAND counterfactual_presets)
