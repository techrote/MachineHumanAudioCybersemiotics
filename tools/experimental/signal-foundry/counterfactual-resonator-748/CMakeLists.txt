cmake_minimum_required(VERSION 3.16)
project(MHACCounterfactualResonator VERSION 0.1.0 LANGUAGES CXX)
add_executable(counterfactual_lab lab.cpp)
target_compile_features(counterfactual_lab PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(counterfactual_lab PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(counterfactual_lab PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME counterfactual_native COMMAND counterfactual_lab)
