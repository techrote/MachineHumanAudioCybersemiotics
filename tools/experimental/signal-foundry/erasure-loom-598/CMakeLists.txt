cmake_minimum_required(VERSION 3.16)
project(ErasureLoom LANGUAGES CXX)
enable_testing()
foreach(target loom_test loom_demo)
  add_executable(${target} ${target}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
endforeach()
add_test(NAME loom_native COMMAND loom_test)
add_executable(loom_live_test live_test.cpp)
target_compile_features(loom_live_test PRIVATE cxx_std_17)
add_test(NAME loom_live_native COMMAND loom_live_test)
add_executable(loom_live_demo live_demo.cpp)
target_compile_features(loom_live_demo PRIVATE cxx_std_17)
foreach(target probe_test probe_demo)
  add_executable(${target} ${target}.cpp)
  target_compile_features(${target} PRIVATE cxx_std_17)
endforeach()
add_test(NAME loom_probe_native COMMAND probe_test)
add_executable(loom_live2_test live2_test.cpp)
target_compile_features(loom_live2_test PRIVATE cxx_std_17)
add_test(NAME loom_live2_native COMMAND loom_live2_test)
add_executable(loom_live2_demo live2_demo.cpp)
target_compile_features(loom_live2_demo PRIVATE cxx_std_17)
