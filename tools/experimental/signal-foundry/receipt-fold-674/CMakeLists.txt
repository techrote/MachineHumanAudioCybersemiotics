cmake_minimum_required(VERSION 3.15)
project(MHACReceiptFold LANGUAGES CXX)
add_executable(receipt_fold_lab lab.cpp)
target_compile_features(receipt_fold_lab PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(receipt_fold_lab PRIVATE /W4 /WX)
else()
  target_compile_options(receipt_fold_lab PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME receipt_fold COMMAND receipt_fold_lab)
add_executable(receipt_fold_live_test live_test.cpp)
target_compile_features(receipt_fold_live_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(receipt_fold_live_test PRIVATE /W4 /WX)
else()
  target_compile_options(receipt_fold_live_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME receipt_fold_live COMMAND receipt_fold_live_test)
