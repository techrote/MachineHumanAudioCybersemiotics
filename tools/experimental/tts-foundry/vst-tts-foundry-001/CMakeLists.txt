function(mhac_strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()

option(MHAC_BUILD_VST3 "Build LOGOS and ARTICULUM VST3 modules against exact SDK 3.8.0" OFF)
if(MHAC_BUILD_VST3)
  set(MHAC_VST3_SDK "" CACHE PATH "Clean recursive VST3 SDK 3.8.0 checkout")
  find_package(Git REQUIRED)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" rev-parse HEAD OUTPUT_VARIABLE sdk_head OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
  if(NOT rc EQUAL 0 OR NOT sdk_head STREQUAL "9fad9770f2ae8542ab1a548a68c1ad1ac690abe0")
    message(FATAL_ERROR "SDK must be exact VST3 3.8.0 build 66")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" status --porcelain --untracked-files=all --ignore-submodules=none OUTPUT_VARIABLE dirty RESULT_VARIABLE rc)
  if(NOT rc EQUAL 0 OR dirty)
    message(FATAL_ERROR "SDK checkout must be recursively clean")
  endif()
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
  add_subdirectory("${MHAC_VST3_SDK}" "${CMAKE_BINARY_DIR}/sdk")
  set(SDK_ROOT "${MHAC_VST3_SDK}")
  set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
  smtg_setup_platform_toolset()
  smtg_setup_symbol_visibility()
  smtg_add_vst3plugin(MHACLogos logos_vst3.cpp)
  target_include_directories(MHACLogos PRIVATE "${SDK_ROOT}")
  target_link_libraries(MHACLogos PRIVATE sdk)
  smtg_target_add_plugin_resources(MHACLogos RESOURCES "${SDK_ROOT}/LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/SDK-LOCK.txt")
  smtg_add_vst3plugin(MHACArticulum articulum_vst3.cpp)
  target_include_directories(MHACArticulum PRIVATE "${SDK_ROOT}")
  target_link_libraries(MHACArticulum PRIVATE sdk)
  smtg_target_add_plugin_resources(MHACArticulum RESOURCES "${SDK_ROOT}/LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/SDK-LOCK.txt")
  if(APPLE)
    smtg_target_set_bundle(MHACLogos BUNDLE_IDENTIFIER "org.mhac.logos" COMPANY_NAME "MHAC")
    smtg_target_set_bundle(MHACArticulum BUNDLE_IDENTIFIER "org.mhac.articulum" COMPANY_NAME "MHAC")
  endif()
  if(WIN32)
    set(MHAC_MODULE_SOURCE "${SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp")
  elseif(UNIX AND NOT APPLE)
    set(MHAC_MODULE_SOURCE "${SDK_ROOT}/public.sdk/source/vst/hosting/module_linux.cpp")
  endif()
  if(MHAC_MODULE_SOURCE)
    add_executable(mhac_tts_sdk_probe tts_sdk_probe.cpp "${MHAC_MODULE_SOURCE}" "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_include_directories(mhac_tts_sdk_probe PRIVATE "${SDK_ROOT}")
    target_link_libraries(mhac_tts_sdk_probe PRIVATE sdk_hosting Threads::Threads ${CMAKE_DL_LIBS})
    if(WIN32)
      target_compile_definitions(mhac_tts_sdk_probe PRIVATE NOMINMAX)
    endif()
  endif()
endif()
