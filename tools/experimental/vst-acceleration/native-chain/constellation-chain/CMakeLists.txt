cmake_minimum_required(VERSION 3.16)
project(mhac_constellation_chain LANGUAGES CXX)
foreach(dep ORBIT BANK SHUTTLE)
  set(${dep}_DIR "" CACHE PATH "Pinned native ${dep} header directory")
  if(NOT IS_DIRECTORY "${${dep}_DIR}")
    message(FATAL_ERROR "Set ${dep}_DIR to the verified pinned header directory")
  endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_sources.py" "${ORBIT_DIR}" "${BANK_DIR}" "${SHUTTLE_DIR}" RESULT_VARIABLE verified)
if(NOT verified EQUAL 0)
  message(FATAL_ERROR "Pinned native source verification failed")
endif()
add_executable(constellation_chain constellation_chain.cpp)
target_compile_features(constellation_chain PRIVATE cxx_std_17)
target_include_directories(constellation_chain PRIVATE "${ORBIT_DIR}" "${BANK_DIR}" "${SHUTTLE_DIR}")
if(MSVC)
  target_compile_options(constellation_chain PRIVATE /W4 /fp:precise)
else()
  target_compile_options(constellation_chain PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME constellation_chain COMMAND constellation_chain)
