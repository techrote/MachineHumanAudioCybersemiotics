cmake_minimum_required(VERSION 3.16)
project(MHACPalimpsestShuttle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(shuttle_test shuttle_test.cpp)
add_executable(shuttle_wav shuttle_wav.cpp)
foreach(target shuttle_test shuttle_wav)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endforeach()
enable_testing()
add_test(NAME shuttle_regression COMMAND shuttle_test)
