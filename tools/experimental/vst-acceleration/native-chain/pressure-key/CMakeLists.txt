cmake_minimum_required(VERSION 3.16)
project(mhac_pressure_key LANGUAGES CXX)
set(DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../Dependencies" CACHE PATH "Pinned native headers")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_sources.py" "${DEPENDENCY_ROOT}" RESULT_VARIABLE verified)
if(NOT verified EQUAL 0)
  message(FATAL_ERROR "Pinned native source verification failed")
endif()
add_executable(pressure_key pressure_key.cpp)
target_compile_features(pressure_key PRIVATE cxx_std_17)
target_include_directories(pressure_key PRIVATE "${DEPENDENCY_ROOT}/Orbit" "${DEPENDENCY_ROOT}/Bank" "${DEPENDENCY_ROOT}/Shuttle")
if(MSVC)
  target_compile_options(pressure_key PRIVATE /W4 /fp:precise)
else()
  target_compile_options(pressure_key PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME pressure_key COMMAND pressure_key)
