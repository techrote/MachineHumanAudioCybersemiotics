if(MHAC_VST3_SDK)
    if(NOT WIN32)
        message(FATAL_ERROR "Combined SDK host probe currently targets Windows")
    endif()
    find_package(Git REQUIRED)
    set(parts base cmake doc pluginterfaces public.sdk tutorials vstgui4)
    set(pins 3d2e82f8e6bff59c1d8b7a27491a29c2286b5206 de6e54eeaaab35b7145f5c32c279b5e892146e04 6d4737c9e70750056e731d88d49aa06eefc8a1a4 31d6eeba6daaa3e2a8bfbe3e7a90ca0b7fbfbc1c a3911a4615dabbfdfd9d181ee26b05c70c289a95 33b73dfbb87f3fde3bce8c0a10cae934dc66ad34 76823bdbe286e4bdb9f79ab8986af5ce7202336c)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" rev-parse HEAD OUTPUT_VARIABLE head OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR NOT head STREQUAL "9fad9770f2ae8542ab1a548a68c1ad1ac690abe0")
        message(FATAL_ERROR "Wrong SDK root")
    endif()
    foreach(i RANGE 0 6)
        list(GET parts ${i} part)
        list(GET pins ${i} pin)
        execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}/${part}" rev-parse HEAD OUTPUT_VARIABLE got OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
        if(NOT rc EQUAL 0 OR NOT got STREQUAL pin)
            message(FATAL_ERROR "Wrong SDK gitlink ${part}")
        endif()
    endforeach()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MHAC_VST3_SDK}" status --porcelain --untracked-files=all --ignore-submodules=none OUTPUT_VARIABLE dirty RESULT_VARIABLE rc)
    if(NOT rc EQUAL 0 OR dirty)
        message(FATAL_ERROR "SDK must be clean")
    endif()
    set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
    add_subdirectory("${MHAC_VST3_SDK}" "${CMAKE_BINARY_DIR}/sdk")
    set(SDK_ROOT "${MHAC_VST3_SDK}")
    set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
    smtg_setup_platform_toolset()
    smtg_setup_symbol_visibility()
    add_executable(mhac_state_probe sdk_state_probe.cpp
        "${SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp"
        "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
    target_compile_definitions(mhac_state_probe PRIVATE NOMINMAX)
    target_include_directories(mhac_state_probe PRIVATE "${SDK_ROOT}")
    target_link_libraries(mhac_state_probe PRIVATE sdk_hosting Threads::Threads)
endif()
